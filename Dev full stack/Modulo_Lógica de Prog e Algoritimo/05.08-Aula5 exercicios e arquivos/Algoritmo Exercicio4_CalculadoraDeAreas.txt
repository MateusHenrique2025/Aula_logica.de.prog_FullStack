Algoritmo "Exercicio4_CalculadoraDeAreas"

// Leia uma opção para calcular a área: 1 – Quadrado (lado × lado);
// 2 – Retângulo (base × altura); 3 – Triângulo ((base × altura) / 2);
// 4 – Círculo (PI × raio², use PI = 3,14).
// Caso o usuário informe uma opção diferente de 1, 2, 3 ou 4,
// exiba "Opção inválida".

Var
// Seção de Declarações das variáveis
   opcao : inteiro
   lado, base, altura, raio, area : real

Inicio
// Seção de comandos, procedimentos, funções, operadores

   pi <- 3.14

   //Calculadora de areas
   escreval("1 – Quadrado")
   escreval("2 – Retângulo")
   escreval("3 – Triângulo")
   escreval("4 – Círculo")
   escreval("")
   escreval("Digite a opção desejada: ")
   leia(opcao)
   escreval("")

   // Estrutura de múltipla escolha para tratar cada opção
   escolha opcao
      caso 1
         escreval("Digite o valor do lado do quadrado: ")
         leia(lado)
         area <- lado * lado
         escreval("A área do quadrado é: ", area:1:2)

      caso 2
         escreval("Digite a base do retângulo: ")
         leia(base)
         escreval("Digite a altura do retângulo: ")
         leia(altura)
         area <- base * altura
         escreval("A área do retângulo é: ", area:1:2)

      caso 3
         escreval("Digite a base do triângulo: ")
         leia(base)
         escreval("Digite a altura do triângulo: ")
         leia(altura)
         area <- (base * altura) / 2
         escreval("A área do triângulo é: ", area:1:2)

      caso 4
         escreval("Digite o raio do círculo: ")
         leia(raio)
         area <- pi * (raio * raio)
         escreval("A área do círculo é: ", area:1:2)

      outrocaso
         escreval("Opção inválida")
   fimescolha
